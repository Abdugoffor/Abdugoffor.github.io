CAP Theorem

Distributed sistema bir vaqtning o'zida faqat 2 ta xususiyatni to'liq ta'minlay oladi:

C — Consistency
Barcha node'lar bir xil ma'lumotni qaytaradi. Yozish darhol hammaga ko'rinadi.
A — Availability
Har bir so'rov javob oladi (xato bo'lmaydi), lekin javob eski bo'lishi mumkin.
P — Partition tolerance
Tarmoq uzilsa ham sistema ishlashda davom etadi. Real dunyoda bu majburiy.
Amaliy qaror: Network partition har doim bo'ladi → P doimo kerak. Demak siz CP yoki AP tanlaysiz.
CP misol: PostgreSQL, MongoDB (strong consistency).
AP misol: Cassandra, DynamoDB, Redis cluster (eventual consistency).

func CreatePayment(ctx context.Context, key string, amount int64) error {
    // 1. Redis'da key bor-yo'qligini tekshir
    exists, err := redis.SetNX(ctx, "idem:"+key, "processing", 24*time.Hour).Result()
    if err != nil { return err }

    if !exists {
        // Key bor — oldingi natijani qaytar
        result, _ := redis.Get(ctx, "idem:result:"+key).Result()
        return json.Unmarshal([]byte(result), &response)
    }

    // 2. Birinchi marta — to'lovni amalga oshir
    payment, err := processPayment(ctx, amount)

    // 3. Natijani saqla
    data, _ := json.Marshal(payment)
    redis.Set(ctx, "idem:result:"+key, data, 24*time.Hour)

    return nil
}

Retry Strategy — Exponential Backoff

Tarmoq xatosi bo'lsa darhol qayta urinmaslik kerak — sistema'ga zarba beradi.

func retryWithBackoff(ctx context.Context, op func() error) error {
    delays := []time.Duration{
        100*time.Millisecond,   // 1-urinish
        500*time.Millisecond,   // 2-urinish
        2*time.Second,          // 3-urinish
        5*time.Second,          // 4-urinish
    }

    for i, delay := range delays {
        err := op()
        if err == nil { return nil }

        // Retriable bo'lmasa darhol chiqib ket
        if !isRetriable(err) { return err }

        // Jitter qo'shamiz — barcha clientlar bir vaqtda urmasin
        jitter := time.Duration(rand.Int63n(100)) * time.Millisecond

        select {
        case <-time.After(delay + jitter):
        case <-ctx.Done(): return ctx.Err()
        }
    }
    return errors.New("max retries exceeded")
}

Distributed Lock

Bir vaqtda bir necha server bir resursga yozmasin.

// Redis-based distributed lock (Redlock pattern)
func withLock(ctx context.Context, key string, ttl time.Duration, fn func() error) error {
    lockKey := "lock:" + key
    token := uuid.New().String()  // kim qulflaganini bilish uchun

    // SET NX — faqat yo'q bo'lsa qo'yadi
    ok, err := redis.SetNX(ctx, lockKey, token, ttl).Result()
    if err != nil { return err }
    if !ok { return errors.New("already locked") }

    defer func() {
        // LUA script — atomik tekshir va o'chir
        script := `if redis.call("GET", KEYS[1]) == ARGV[1] then
            return redis.call("DEL", KEYS[1])
        else return 0 end`
        redis.Eval(ctx, script, []string{lockKey}, token)
    }()

    return fn()
}

Split Brain

Cluster ikkiga bo'linib qoladi va har biri o'zini "master" deb hisoblaydi.

Normal cluster: Network partition: ┌─────────────────┐ ┌──────┐ ✗ ┌──────┐ │ N1 N2 N3 │ │ N1 │ │ N2 │ │ master: N1 │ → │ N3 │ │ N2 │ │ │ │master│ │master│ ← split brain! └─────────────────┘ └──────┘ └──────┘
Yechim — Quorum: Master tanlash uchun N/2 + 1 ovoz kerak. 3 node'li clusterda 2 ovoz. Shunda ozchilik tomon master tanlay olmaydi.

Real interview savol — to'liq javob

Savol: Foydalanuvchi "Pay" tugmasini 3 marta bosdi. Sizning mikroservisingiz aniq 1 marta to'lov qilishini qanday ta'minlaysiz?

Javob — 4 qatlamli himoya

  1. Frontend: Tugma bosilgandan keyin disabled qilinadi, debouncing 500ms.
  2. Idempotency-Key: Client har to'lov uchun UUID generate qiladi va header'da yuboradi.
  3. Redis cache: Backend key'ni SETNX bilan tekshiradi — 24 soat saqlaydi.
  4. DB constraint: payments jadvalida UNIQUE(idempotency_key) — final himoya.

        },

// Middleware
func metricsMiddleware(next http.Handler) http.Handler {
    return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
        start := time.Now()
        rw := &responseWriter{ResponseWriter: w, status: 200}

        next.ServeHTTP(rw, r)

        duration := time.Since(start).Seconds()
        httpRequests.WithLabelValues(r.Method, r.URL.Path, fmt.Sprint(rw.status)).Inc()
        httpDuration.WithLabelValues(r.Method, r.URL.Path).Observe(duration)
    })
}

JWT + Refresh Token Pattern

Muhim: Access token qisqa umrli (15 min), Refresh token uzun (7-30 kun). Refresh token DB'da saqlanadi va rotation qilinadi. 
import "github.com/golang-jwt/jwt/v5"

type Claims struct {
    UserID int64  `json:"uid"`
    Role   string `json:"role"`
    jwt.RegisteredClaims
}

// Access token (short-lived)
func generateAccess(userID int64, role string) (string, error) {
    claims := Claims{
        UserID: userID,
        Role:   role,
        RegisteredClaims: jwt.RegisteredClaims{
            ExpiresAt: jwt.NewNumericDate(time.Now().Add(15 * time.Minute)),
            IssuedAt:  jwt.NewNumericDate(time.Now()),
            Issuer:    "myapp",
        },
    }
    token := jwt.NewWithClaims(jwt.SigningMethodHS256, claims)
    return token.SignedString(secretKey)
}

// Refresh token (long-lived, DB'da saqlanadi)
func generateRefresh(userID int64) (string, error) {
    token := uuid.New().String()
    hash := sha256.Sum256([]byte(token))

    _, err := db.Exec(ctx, `
        INSERT INTO refresh_tokens (user_id, token_hash, expires_at)
        VALUES ($1, $2, $3)`,
        userID, hash[:], time.Now().Add(7*24*time.Hour),
    )
    return token, err
}

// Refresh endpoint — token rotation
func refreshHandler(w http.ResponseWriter, r *http.Request) {
    oldToken := getRefreshFromCookie(r)
    hash := sha256.Sum256([]byte(oldToken))

    // 1. Eski tokenni topish
    var userID int64
    var used bool
    err := db.QueryRow(ctx, `
        SELECT user_id, used FROM refresh_tokens
        WHERE token_hash = $1 AND expires_at > NOW()`,
        hash[:],
    ).Scan(&userID, &used)

    if err != nil || used {
        // ⚠ used=true bo'lsa — token o'g'irlangan!
        // Bu user'ning barcha refresh tokenlarini bekor qiling
        db.Exec(ctx, `UPDATE refresh_tokens SET revoked=true WHERE user_id=$1`, userID)
        http.Error(w, "unauthorized", 401)
        return

func RequirePermission(resource, action string) gin.HandlerFunc {
    return func(c *gin.Context) {
        userID := c.GetInt64("user_id")

        var count int
        db.QueryRow(ctx, `

            JOIN role_permissions rp ON ur.role_id = rp.role_id
            JOIN permissions p ON rp.permission_id = p.id

            userID, resource, action,

            c.AbortWithStatusJSON(403, gin.H{"error": "forbidden"})

    }

// Server: connection manager
type Hub struct {
    clients    map[int64]*Client  // user_id → connection
    mu         sync.RWMutex
    redis      *redis.Client
    serverID   string
}

func (h *Hub) Register(c *Client) {
    h.mu.Lock()
    h.clients[c.UserID] = c
    h.mu.Unlock()

    // Redis'da user qaysi serverda — saqlash
    h.redis.Set(ctx, fmt.Sprintf("user:%d:server", c.UserID), h.serverID, 5*time.Minute)

    // User'ning shaxsiy kanaliga subscribe
    h.subscribeUser(c.UserID)
}

// Xabar yuborish
func (h *Hub) SendToUser(ctx context.Context, userID int64, msg []byte) error {
    // 1. Lokal serverda bormi?
    h.mu.RLock()
    client, ok := h.clients[userID]
    h.mu.RUnlock()

    if ok {
        client.Send <- msg  // to'g'ridan-to'g'ri yuborish
        return nil
    }

    // 2. Boshqa serverda — Redis Pub/Sub orqali
    channel := fmt.Sprintf("user:%d", userID)
    return h.redis.Publish(ctx, channel, msg).Err()
}

// Subscriber — boshqa serverdan keladigan xabarlar
func (h *Hub) subscribeUser(userID int64) {
    sub := h.redis.Subscribe(ctx, fmt.Sprintf("user:%d", userID))
    go func() {
        for msg := range sub.Channel() {
            h.mu.RLock()
            client, ok := h.clients[userID]
            h.mu.RUnlock()
            if ok {
                client.Send <- []byte(msg.Payload)
            }
        }
    }()
}

    };

    this.ws.onclose = () => {
      if (this.attempts >= this.maxAttempts) return;

      // Exponential backoff + jitter
      const delay = Math.min(1000 * 2 ** this.attempts, 30000)
                  + Math.random() * 1000;
      this.attempts++;
      setTimeout(() => this.connect(), delay);
    };
  }
}

func (c *Client) writePump() {
    for msg := range c.Send {
        c.Conn.SetWriteDeadline(time.Now().Add(10 * time.Second))
        if err := c.Conn.WriteMessage(websocket.TextMessage, msg); err != nil {
            return
        }

}

        // Optional: connection'ni uzish

    }

type OrderSaga struct {
    orderSvc     OrderService
    inventorySvc InventoryService
    paymentSvc   PaymentService
}

func (s *OrderSaga) Execute(ctx context.Context, req CreateOrderRequest) error {
    // Compensating actions stack'i
    var compensations []func() error

    rollback := func() {
        for i := len(compensations) - 1; i >= 0; i-- {
            if err := compensations[i](); err != nil {
                log.Error("compensation failed", "err", err)
                // Manual intervention queue'ga qo'yish
            }
        }
    }

    // 1. Order yaratish
    order, err := s.orderSvc.Create(ctx, req)
    if err != nil { return err }
    compensations = append(compensations, func() error {
        return s.orderSvc.Cancel(ctx, order.ID)
    })

    // 2. Inventory reserve
    if err := s.inventorySvc.Reserve(ctx, order.Items); err != nil {
        rollback()
        return err
    }
    compensations = append(compensations, func() error {
        return s.inventorySvc.Release(ctx, order.Items)
    })

    // 3. Payment
    if err := s.paymentSvc.Charge(ctx, order.UserID, order.Total); err != nil {
        rollback()
        return err
    }

    // 4. Confirm
    return s.orderSvc.Confirm(ctx, order.ID)
}

func (s *OrderService) Create(ctx context.Context, req Request) error {
    return withTx(ctx, s.db, func(tx *sql.Tx) error {
        // 1. Order'ni saqlash
        order := domain.NewOrder(req)
        if _, err := tx.Exec("INSERT INTO orders ...", ...); err != nil {
            return err
        }

        // 2. Event'ni outbox'ga yozish — bir tx ichida
        payload, _ := json.Marshal(domain.OrderCreatedEvent{ID: order.ID})
        _, err := tx.Exec(`
            INSERT INTO outbox (aggregate_type, aggregate_id, event_type, payload)
            VALUES ($1, $2, $3, $4)`,
            "order", order.ID, "OrderCreated", payload,
        )
        return err
    })
}

// Outbox publisher (alohida goroutine yoki worker)
func (p *OutboxPublisher) Run(ctx context.Context) {
    ticker := time.NewTicker(500 * time.Millisecond)
    for {
        select {
        case <-ctx.Done(): return
        case <-ticker.C:
            rows, _ := p.db.Query(`
                SELECT id, event_type, payload FROM outbox
                WHERE published_at IS NULL
                ORDER BY id LIMIT 100`)

            for rows.Next() {
                var id int64; var et string; var pl []byte
                rows.Scan(&id, &et, &pl)

                if err := p.kafka.Publish(et, pl); err == nil {
                    p.db.Exec("UPDATE outbox SET published_at=NOW() WHERE id=$1", id)
                }
            }
        }
    }
}
