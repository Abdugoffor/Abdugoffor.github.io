    "fmt"
    "sync"

type Cache[K comparable,V any]struct{mu sync.RWMutex;data map[K]V}
func NewCache[K comparable,V any]()*Cache[K,V]{return &Cache[K,V]{data:make(map[K]V)}}
func(c *Cache[K,V])Set(k K,v V){c.mu.Lock();defer c.mu.Unlock();c.data[k]=v}

        wg.Add(1)
        go func(){defer wg.Done();c.Set(id,fmt.Sprintf("User-%d",id))}()
    }
    wg.Wait()
    val,ok:=c.Get(42)
    fmt.Println(val,ok)
}

package main

import (
    "bytes"
    "crypto/rand"
    "encoding/hex"
    "fmt"
    "io"
    "net/http"
    "net/http/httptest"
    "os"
    "path/filepath"
)
const maxSize int64=5<<20

type FileService struct{Dir string}
func(s *FileService)Save(r io.Reader)(string,error){
    if err:=os.MkdirAll(s.Dir,0700);err!=nil{return "",err}
    id:=make([]byte,16)
    if _,err:=rand.Read(id);err!=nil{return "",err}
    name:=hex.EncodeToString(id)+".bin"
    path:=filepath.Join(s.Dir,name)
    f,err:=os.OpenFile(path,os.O_CREATE|os.O_EXCL|os.O_WRONLY,0600)
    if err!=nil{return "",err}
    n,copyErr:=io.Copy(f,io.LimitReader(r,maxSize+1))
    closeErr:=f.Close()
    if copyErr!=nil||closeErr!=nil||n>maxSize{
        _=os.Remove(path)
        if copyErr!=nil{return "",copyErr}
        if closeErr!=nil{return "",closeErr}
        return "",fmt.Errorf("file too large")
    }
    return name,nil
}

func main(){
    dir,err:=os.MkdirTemp("","uploads-*")
    if err!=nil{panic(err)}
    defer os.RemoveAll(dir)
    s:=&FileService{Dir:dir}
    h:=http.HandlerFunc(func(w http.ResponseWriter,r *http.Request){
        r.Body=http.MaxBytesReader(w,r.Body,maxSize+1)
        defer r.Body.Close()
        _,err:=s.Save(r.Body)
        if err!=nil{http.Error(w,"upload failed",400);return}
        fmt.Fprintln(w,"Saved")
    })
    rec:=httptest.NewRecorder()
    h.ServeHTTP(rec,httptest.NewRequest("POST","/upload",bytes.NewBufferString("Hello Go")))
    fmt.Println(rec.Code)
}

Streaming CSV + bounded worker pool

3-misol
Go
package main

import (
    "encoding/csv"
    "fmt"
    "io"
    "strconv"
    "strings"
    "sync"
)

type Order struct { ID, Amount int }

func main() {
    data := "1,100\n2,200\n3,300\n4,400\n5,500\n"
    reader := csv.NewReader(strings.NewReader(data))
    jobs := make(chan Order, 3)
    results := make(chan int, 3)
    readErrors := make(chan error, 1)
    var wg sync.WaitGroup

    for i := 0; i < 4; i++ {
        wg.Add(1)
        go func() {
            defer wg.Done()
            for order := range jobs { results <- order.Amount }
        }()
    }
    go func() {
        defer close(jobs)
        for {
            row, err := reader.Read()
            if err == io.EOF { readErrors <- nil; return }
            if err != nil { readErrors <- err; return }
            if len(row) != 2 {
                readErrors <- fmt.Errorf("invalid CSV row")
                return
            }
            id, err := strconv.Atoi(row[0])
            if err != nil { readErrors <- err; return }
            amount, err := strconv.Atoi(row[1])
            if err != nil { readErrors <- err; return }
            jobs <- Order{ID: id, Amount: amount}
        }
    }()
    go func() { wg.Wait(); close(results) }()

    total := 0
    for amount := range results { total += amount }
    if err := <-readErrors; err != nil { panic(err) }
    fmt.Println("Total:", total)
}
Konsoldagi natija
Total: 1500

Nima uchun? Jobs bufferi cheklangan, shuning uchun millionlab qatorlarni RAMda to‘plamaydi. Workerlar results kanaliga yozadi; main ulardan o‘qiydi. Production: ctx, batch insert, qator limitlari va xatolarni kuzatish.

Context bilan retry: vaqtinchalik xatolar

package main

import (
    "context"
    "errors"
    "fmt"
    "time"
)

var ErrTemporary = errors.New("temporary")

func Retry(ctx context.Context, max int, work func(context.Context) error) error {
    if max < 1 { return fmt.Errorf("invalid attempts") }
    for i := 0; i < max; i++ {
        if err := ctx.Err(); err != nil { return err }
        err := work(ctx)
        if err == nil { return nil }
        if !errors.Is(err, ErrTemporary) || i == max-1 { return err }
        timer := time.NewTimer(time.Duration(i+1) * 10 * time.Millisecond)
        select {
        case <-ctx.Done():
            timer.Stop()
            return ctx.Err()
        case <-timer.C:
        }
    }
    return nil
}

func main() {
    ctx, cancel := context.WithTimeout(context.Background(), time.Second)
    defer cancel()
    attempt := 0
    err := Retry(ctx, 3, func(context.Context) error {
        attempt++
        fmt.Println("Attempt:", attempt)
        if attempt < 3 { return ErrTemporary }
        return nil
    })
    fmt.Println("Error:", err)
}

package main

import (
    "encoding/csv"
    "fmt"
    "io"
    "os"
    "strconv"
)

type Order struct { ID, Amount int }
type OrderService struct{}

func (s *OrderService) Export(orders []Order) *io.PipeReader {
    reader, writer := io.Pipe()
    go func() {
        csvWriter := csv.NewWriter(writer)
        for _, o := range orders {
            err := csvWriter.Write([]string{
                strconv.Itoa(o.ID), strconv.Itoa(o.Amount),
            })
            if err != nil { _ = writer.CloseWithError(err); return }
        }
        csvWriter.Flush()
        if err := csvWriter.Error(); err != nil {
            _ = writer.CloseWithError(err)
            return
        }
        _ = writer.Close()
    }()
    return reader
}

func main() {
    s := &OrderService{}
    reader := s.Export([]Order{{1, 100}, {2, 200}})
    defer reader.Close()
    if _, err := io.Copy(os.Stdout, reader); err != nil { panic(err) }
    fmt.Println("Export completed")
}

2,200
Export completed

package main

import (
    "bytes"
    "encoding/json"
    "errors"
    "fmt"
    "io"
    "mime"
    "net/http"
    "net/http/httptest"
    "strings"
)

type CreateProduct struct {
    Name string `json:"name"`
    Price int `json:"price"`
}

type Product struct {
    ID int `json:"id"`
    Name string `json:"name"`
    Price int `json:"price"`
}

func CreateHandler(w http.ResponseWriter, r *http.Request) {
    if r.Method != http.MethodPost {
        http.Error(w, "method not allowed", 405)
        return
    }
    mediaType, _, err := mime.ParseMediaType(r.Header.Get("Content-Type"))
    if err != nil || mediaType != "application/json" {
        http.Error(w, "JSON required", 415)
        return
    }
    r.Body = http.MaxBytesReader(w, r.Body, 1<<20)
    defer r.Body.Close()
    dec := json.NewDecoder(r.Body)
    dec.DisallowUnknownFields()
    var req CreateProduct
    if err := dec.Decode(&req); err != nil {
        http.Error(w, "invalid JSON", 400)
        return
    }
    var extra any
    if err := dec.Decode(&extra); !errors.Is(err, io.EOF) {
        http.Error(w, "extra JSON", 400)
        return
    }
    if strings.TrimSpace(req.Name) == "" || req.Price <= 0 {
        http.Error(w, "validation failed", 422)
        return
    }
    w.Header().Set("Content-Type", "application/json")
    w.WriteHeader(http.StatusCreated)
    _ = json.NewEncoder(w).Encode(Product{1, req.Name, req.Price})
}

func main() {
    req := httptest.NewRequest("POST", "/products",
        bytes.NewBufferString(`{"name":"Laptop","price":1500}`))
    req.Header.Set("Content-Type", "application/json")
    rec := httptest.NewRecorder()
    CreateHandler(rec, req)
    fmt.Println("Status:", rec.Code)
    fmt.Println("Body:", strings.TrimSpace(rec.Body.String()))
}

Body: {"id":1,"name":"Laptop","price":1500}

package main

import (
    "context"
    "errors"
    "fmt"
    "time"
)

type DBError struct { Code string }
func (e *DBError) Error() string { return "database error: " + e.Code }

func retryable(err error) bool {
    var e *DBError
    if !errors.As(err, &e) { return false }
    return e.Code == "40001" || e.Code == "40P01"
}

func WithRetry(ctx context.Context, attempts int, work func(context.Context) error) error {
    if attempts < 1 { return errors.New("attempts must be positive") }
    for i := 1; i <= attempts; i++ {
        if err := ctx.Err(); err != nil { return err }
        err := work(ctx)
        if err == nil { return nil }
        if !retryable(err) || i == attempts { return err }
        timer := time.NewTimer(time.Duration(i) * 10 * time.Millisecond)
        select {
        case <-ctx.Done():
            timer.Stop()
            return ctx.Err()
        case <-timer.C:
        }
    }
    return nil
}

func main() {
    ctx, cancel := context.WithTimeout(context.Background(), time.Second)
    defer cancel()
    attempt := 0
    err := WithRetry(ctx, 3, func(context.Context) error {
        attempt++
        fmt.Println("Transaction:", attempt)
        if attempt < 3 { return &DBError{Code: "40001"} }
        return nil
    })
    fmt.Println("Result:", err)
}

type Order struct { ID string; Amount int }
type Repository interface {
    Create(context.Context, Order) (bool, error)
}

type MemoryRepo struct {
    mu sync.Mutex
    stock int
    saved map[string]Order
}

func (r *MemoryRepo) Create(ctx context.Context, o Order) (bool, error) {
    if err := ctx.Err(); err != nil { return false, err }
    r.mu.Lock()
    defer r.mu.Unlock()
    if _, ok := r.saved[o.ID]; ok { return false, nil }
    if r.stock <= 0 { return false, ErrStock }
    r.stock--
    r.saved[o.ID] = o
    return true, nil
}

type Service struct { repo Repository }
func (s Service) Create(ctx context.Context, o Order) (bool, error) {
    if o.ID == "" || o.Amount <= 0 { return false, errors.New("invalid order") }
    return s.repo.Create(ctx, o)
}

func main() {

    service := Service{repo: repo}
    for _, o := range []Order{{"a",100},{"a",100},{"b",200},{"c",300}} {
        created, err := service.Create(context.Background(), o)
        fmt.Println(o.ID, created, err)

}

package main

import (
    "fmt"
    "sync"
    "sync/atomic"
)

type Inventory struct {
    mu sync.Mutex
    stock int
    seen map[string]bool
}

func (i *Inventory) Reserve(key string) bool {
    i.mu.Lock()
    defer i.mu.Unlock()
    if i.seen[key] { return false } // Dublikat order
    if i.stock == 0 { return false }
    i.seen[key] = true
    i.stock--
    return true
}

func (i *Inventory) Stock() int {
    i.mu.Lock()
    defer i.mu.Unlock()
    return i.stock
}

func main() {
    inventory := &Inventory{stock: 10, seen: make(map[string]bool)}
    // Bitta haqiqiy rezervatsiyani oldindan yaratamiz.
    inventory.Reserve("order-1")
    var wg sync.WaitGroup
    var success atomic.Int64
    success.Store(1)
    for id := 1; id <= 100; id++ {
        wg.Add(1)
        go func(orderID int) {
            defer wg.Done()
            if inventory.Reserve(fmt.Sprintf("order-%d", orderID)) {
                success.Add(1)
            }
        }(id)
    }
    wg.Wait()
    fmt.Println("Success:", success.Load())
    fmt.Println("Stock:", inventory.Stock())
    fmt.Println("Duplicate accepted:", inventory.Reserve("order-1"))
}
